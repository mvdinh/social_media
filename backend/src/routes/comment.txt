import express from 'express';
import Comment from '../models/Comment.js';
import { getContract } from '../config/contract.js';

const router = express.Router();

// ==================== ADD COMMENT ====================
router.post('/', async (req, res) => {
  try {
    const { postId, author, contentHash, mediaHash } = req.body;

    // 1️⃣ Validation
    if (!postId || !author || !contentHash) {
      return res.status(400).json({
        success: false,
        error: 'Missing required fields: postId, author, contentHash',
      });
    }

    // 2️⃣ Get contract instance
    const contract = getContract();
    console.log(`⏳ Adding comment to post ${postId}...`);

    // 3️⃣ Call smart contract - Use correct function based on mediaHash
    let tx;
    if (mediaHash && mediaHash.trim() !== '') {
      // Use addCommentWithMedia if mediaHash exists
      console.log('📸 Adding comment with media...');
      tx = await contract.addCommentWithMedia(postId, contentHash, mediaHash);
    } else {
      // Use addComment for text-only
      console.log('📝 Adding text comment...');
      tx = await contract.addComment(postId, contentHash);
    }
    
    console.log('📤 Transaction sent:', tx.hash);

    // 4️⃣ Wait for confirmation
    const receipt = await tx.wait();
    console.log('✅ Transaction confirmed:', receipt.hash);

    // 5️⃣ Parse event from logs
    const parsedLogs = receipt.logs
      .map((log) => {
        try {
          return contract.interface.parseLog(log);
        } catch {
          return null;
        }
      })
      .filter(Boolean);

    const commentAddedEvent = parsedLogs.find(
      (event) => event && event.name === 'CommentAdded'
    );

    if (!commentAddedEvent) {
      console.warn('⚠️ CommentAdded event not found, but transaction succeeded');
    }

    console.log('📢 CommentAdded event:', commentAddedEvent?.args);

    // 6️⃣ Verify data if event exists
    if (commentAddedEvent) {
      const {
        postId: eventPostId,
        author: eventAuthor,
        contentHash: eventContentHash,
      } = commentAddedEvent.args;

      const matches = {
        postId: Number(eventPostId) === Number(postId),
        author: eventAuthor.toLowerCase() === author.toLowerCase(),
        contentHash: eventContentHash === contentHash,
      };

      if (!Object.values(matches).every(Boolean)) {
        return res.status(400).json({
          success: false,
          error: 'Comment data mismatch between request and blockchain event',
          details: matches,
        });
      }
    }

    console.log('✅ Blockchain verification passed');

    // 7️⃣ Get comment index from blockchain
    const comments = await contract.getComments(postId);
    const commentIndex = comments.length - 1;

    // 8️⃣ Save to database
    const comment = await Comment.create({
      postId: Number(postId),
      commentIndex: commentIndex,
      author: author.toLowerCase(),
      contentHash,
      mediaHash: mediaHash || '',
      timestamp: new Date(),
      txHash: receipt.hash,
      blockNumber: receipt.blockNumber,
      isDeleted: false,
    });

    console.log('💾 Comment saved to database:', comment._id);

    // 9️⃣ Return success response
    res.json({
      success: true,
      comment,
      blockchainData: {
        commentIndex: commentIndex,
        author: commentAddedEvent?.args?.author || author,
        contentHash: commentAddedEvent?.args?.contentHash || contentHash,
        blockNumber: receipt.blockNumber,
        txHash: receipt.hash,
      },
      message: 'Comment verified on blockchain and saved to database',
    });

  } catch (error) {
    console.error('❌ Error adding comment:', error);

    // Handle specific blockchain errors
    if (error.code === 'CALL_EXCEPTION') {
      return res.status(400).json({
        success: false,
        error: 'Smart contract call failed',
        details: error.reason || error.message,
        hint: 'Make sure the post exists and you have the correct postId',
      });
    }

    if (error.code === 'NETWORK_ERROR') {
      return res.status(503).json({
        success: false,
        error: 'Blockchain network error',
        details: error.message,
      });
    }

    if (error.code === 'INSUFFICIENT_FUNDS') {
      return res.status(402).json({
        success: false,
        error: 'Insufficient funds for gas',
        details: error.message,
      });
    }

    res.status(500).json({
      success: false,
      error: 'Failed to add comment',
      details: error.message,
    });
  }
});

// ==================== DELETE COMMENT ====================
router.delete('/:postId/:commentIndex', async (req, res) => {
  try {
    const { postId, commentIndex } = req.params;
    const { author } = req.body;

    if (!author) {
      return res.status(400).json({
        success: false,
        error: 'Missing required field: author',
      });
    }

    // ⚠️ NOTE: Smart contract doesn't have deleteComment function
    // We can only soft delete in database
    const comment = await Comment.findOneAndUpdate(
      { 
        postId: Number(postId), 
        commentIndex: Number(commentIndex),
        author: author.toLowerCase() 
      },
      { isDeleted: true },
      { new: true }
    );

    if (!comment) {
      return res.status(404).json({
        success: false,
        error: 'Comment not found or you are not the author',
      });
    }

    res.json({
      success: true,
      message: 'Comment marked as deleted (soft delete)',
      comment,
    });

  } catch (error) {
    console.error('❌ Error deleting comment:', error);
    res.status(500).json({
      success: false,
      error: 'Failed to delete comment',
      details: error.message,
    });
  }
});

// ==================== GET COMMENTS BY POST ID ====================
router.get('/:postId', async (req, res) => {
  try {
    const { postId } = req.params;
    const { includeDeleted } = req.query;

    // Validate postId
    if (!postId || isNaN(Number(postId))) {
      return res.status(400).json({
        success: false,
        error: 'Invalid postId',
      });
    }

    // Build filter
    const filter = {
      postId: Number(postId),
    };

    // Only include non-deleted comments unless specifically requested
    if (includeDeleted !== 'true') {
      filter.isDeleted = false;
    }

    // Fetch comments from database
    const comments = await Comment.find(filter).sort({ timestamp: 1 });

    res.json({
      success: true,
      comments,
      count: comments.length,
    });

  } catch (error) {
    console.error('❌ Error fetching comments:', error);
    res.status(500).json({
      success: false,
      error: 'Failed to fetch comments',
      details: error.message,
    });
  }
});

// ==================== GET COMMENTS FROM BLOCKCHAIN ====================
router.get('/blockchain/:postId', async (req, res) => {
  try {
    const { postId } = req.params;

    if (!postId || isNaN(Number(postId))) {
      return res.status(400).json({
        success: false,
        error: 'Invalid postId',
      });
    }

    const contract = getContract();
    const blockchainComments = await contract.getComments(postId);

    // Convert BigInt timestamps to numbers
    const formattedComments = blockchainComments.map((comment, index) => ({
      commentIndex: index,
      author: comment.author,
      contentHash: comment.contentHash,
      mediaHash: comment.mediaHash,
      timestamp: Number(comment.timestamp),
    }));

    res.json({
      success: true,
      comments: formattedComments,
      count: formattedComments.length,
      source: 'blockchain',
    });

  } catch (error) {
    console.error('❌ Error fetching blockchain comments:', error);
    res.status(500).json({
      success: false,
      error: 'Failed to fetch comments from blockchain',
      details: error.message,
    });
  }
});

// ==================== SYNC COMMENTS FROM BLOCKCHAIN ====================
router.post('/sync/:postId', async (req, res) => {
  try {
    const { postId } = req.params;

    if (!postId || isNaN(Number(postId))) {
      return res.status(400).json({
        success: false,
        error: 'Invalid postId',
      });
    }

    const contract = getContract();
    const blockchainComments = await contract.getComments(postId);

    let synced = 0;
    let skipped = 0;

    for (let i = 0; i < blockchainComments.length; i++) {
      const bcComment = blockchainComments[i];

      // Check if already exists
      const exists = await Comment.findOne({
        postId: Number(postId),
        commentIndex: i,
      });

      if (exists) {
        skipped++;
        continue;
      }

      // Create new comment
      await Comment.create({
        postId: Number(postId),
        commentIndex: i,
        author: bcComment.author.toLowerCase(),
        contentHash: bcComment.contentHash,
        mediaHash: bcComment.mediaHash || '',
        timestamp: new Date(Number(bcComment.timestamp) * 1000),
        txHash: 'synced',
        blockNumber: 0,
        isDeleted: false,
      });

      synced++;
    }

    res.json({
      success: true,
      message: `Synced ${synced} comments, skipped ${skipped} existing`,
      synced,
      skipped,
      total: blockchainComments.length,
    });

  } catch (error) {
    console.error('❌ Error syncing comments:', error);
    res.status(500).json({
      success: false,
      error: 'Failed to sync comments',
      details: error.message,
    });
  }
});

// ==================== GET ALL COMMENTS (ADMIN) ====================
router.get('/admin/all', async (req, res) => {
  try {
    const { limit = 100, skip = 0, includeDeleted = false } = req.query;

    const filter = includeDeleted === 'true' ? {} : { isDeleted: false };

    const comments = await Comment.find(filter)
      .sort({ timestamp: -1 })
      .limit(Number(limit))
      .skip(Number(skip));

    const total = await Comment.countDocuments(filter);

    res.json({
      success: true,
      comments,
      count: comments.length,
      total,
      pagination: {
        limit: Number(limit),
        skip: Number(skip),
        hasMore: Number(skip) + comments.length < total,
      },
    });

  } catch (error) {
    console.error('❌ Error fetching all comments:', error);
    res.status(500).json({
      success: false,
      error: 'Failed to fetch comments',
      details: error.message,
    });
  }
});

export default router;
