import React, { useEffect, useState } from 'react';
import axios from 'axios';
import { CheckCircle, MoreHorizontal } from 'lucide-react';
import { Post } from '../../types';
import LikeButton from '../LikeButton';
import CommentButton from '../CommentButton';
import ShareButton from '../ShareButton';
import CommentsSection from '../CommentsSection';

interface PostCardProps {
  post: Post;
  user: any;        // nếu bạn có type riêng cho user thì thay thế bằng interface User
  contract: any;
}

const API_URL = import.meta.env.VITE_BACKEND_URL || 'http://localhost:3000';

const PostCard: React.FC<PostCardProps> = ({ post, user, contract }) => {
  const [isLiked, setIsLiked] = useState<boolean>(false);
  const [commentsOpen, setCommentsOpen] = useState<boolean>(false);

  // Xử lý thời gian tạo bài viết (nếu backend có field createdAt)
  const timeAgo = post.createdAt
    ? new Date(post.createdAt).toLocaleDateString('en-US', {
        month: 'short',
        day: 'numeric',
      })
    : 'N/A';

  // Kiểm tra người dùng đã like bài viết hay chưa
  const checkIsLike = async () => {
    try {
      const res = await axios.get(`${API_URL}/likes/check/${post.postId}/${user}`);
      if (res.data?.data?.isLiked) {
        setIsLiked(true);
      }
    } catch (error) {
      console.error('❌ Error checking like status:', error);
    }
  };

  useEffect(() => {
    if (user && post.postId) {
      checkIsLike();
    }
  }, [user, post.postId]);

  return (
    <div className="bg-white p-5 rounded-xl shadow-md border border-gray-100 mb-6">
      {/* Header */}
      <div className="flex items-start justify-between mb-4">
        <div className="flex items-center space-x-3">
          <img
            src={user.profile_picture}
            alt={user.full_name}
            className="w-10 h-10 rounded-full object-cover"
          />
          <div>
            <div className="flex items-center space-x-1">
              <span className="font-bold text-gray-900">{user.full_name}</span>
              {user.is_verified && (
                <CheckCircle className="w-4 h-4 text-blue-500 fill-blue-500" />
              )}
            </div>
            <p className="text-sm text-gray-500">
              @{user.username} · {timeAgo}
            </p>
          </div>
        </div>
        <MoreHorizontal className="w-5 h-5 text-gray-500 cursor-pointer hover:text-gray-700" />
      </div>

      {/* Nội dung bài viết */}
      {post.contentHash && (
        <p className="text-gray-700 mb-4 whitespace-pre-wrap">
          {/* Bạn có thể gọi IPFS để lấy nội dung thực tế nếu muốn */}
          {`📄 Content hash: ${post.contentHash}`}
        </p>
      )}

      {/* Ảnh hoặc video */}
      {post.mediaHashes && post.mediaHashes.length > 0 && (
        <div className="mb-4 rounded-lg overflow-hidden">
          <img
            src={`https://ipfs.io/ipfs/${post.mediaHashes[0]}`}
            alt="Post media"
            className="w-full object-cover max-h-96"
          />
        </div>
      )}

      {/* Nút Like, Comment, Share */}
      <div className="flex items-center space-x-6 text-sm text-gray-500">
        <LikeButton
          postId={post.postId}
          contract={contract}
          user={user}
          countLikes={post.likes}
          isLiked={isLiked}
        />
        <CommentButton
          countComments={post.comments}
          contract={contract}
          user={user}
          onClick={() => setCommentsOpen(!commentsOpen)}
        />
        <ShareButton 
          postId={post.postId} 
          countShares={post.shares}
          contract={contract}
          user={user}
        />
      </div>

      {/* Comment Section */}
      <CommentsSection 
        postId={post.postId} 
        contract={contract}
        user={user}
        isOpen={commentsOpen} 
      />
    </div>
  );
};

export default PostCard;
