// Component nút Share
const ShareButton = ({ postId, countShares, user, contract }) => {
  const [shares, setShares] = useState(countShares);
  const [loading, setLoading] = useState(false);

  const handleShare = async () => {
    if (loading) return;
    
    setLoading(true);
    try {
      // Call API share
      const response = await axios.post(`/api/posts/${postId}/share`, {
        method: 'POST',
      });
      const data = await response.json();
      
      setShares(data.total_shares);
    } catch (error) {
      console.error('Error sharing post:', error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div
      className={`flex items-center space-x-1 cursor-pointer hover:text-green-500 transition-colors ${
        loading ? 'opacity-50' : ''
      }`}
      onClick={handleShare}
    >
      <Share2 className="w-5 h-5" />
      <span>{shares}</span>
    </div>
  );
};
