// Component hiển thị một comment
const CommentItem = ({ comment }) => {
  const timeAgo = new Date(comment.created_at).toLocaleDateString("en-US", {
    month: "short",
    day: "numeric",
  });

  return (
    <div className="flex space-x-3 py-3">
      <img
        src={comment.user.profile_picture}
        alt={comment.user.full_name}
        className="w-8 h-8 rounded-full object-cover flex-shrink-0"
      />
      <div className="flex-1">
        <div className="bg-gray-50 rounded-lg px-3 py-2">
          <div className="flex items-center space-x-1 mb-1">
            <span className="font-semibold text-sm text-gray-900">
              {comment.user.full_name}
            </span>
            {comment.user.is_verified && (
              <CheckCircle className="w-3 h-3 text-blue-500 fill-blue-500" />
            )}
          </div>
          <p className="text-sm text-gray-700">{comment.content}</p>
        </div>
        <div className="flex items-center space-x-4 mt-1 px-3">
          <span className="text-xs text-gray-500">{timeAgo}</span>
          <button className="text-xs text-gray-500 hover:text-blue-500 font-medium">
            Thích
          </button>
          <button className="text-xs text-gray-500 hover:text-blue-500 font-medium">
            Trả lời
          </button>
        </div>
      </div>
    </div>
  );
};
