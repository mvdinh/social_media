import { useState, useEffect } from "react";
import { FriendRequest } from "../../types/friend";
import { FriendRequestList } from "./FriendRequestList";
import axios from "axios";
import { useAuth } from "../../context/AuthContext";
import { connectWallet, getContract } from "../../utils/wallet";
import { ethers } from "ethers";

export function FriendRequestsPage() {
  const { address } = useAuth();
  const [requests, setRequests] = useState<FriendRequest[]>([]);
  const [notification, setNotification] = useState<string>("");

  const showNotification = (message: string) => {
    setNotification(message);
    setTimeout(() => setNotification(""), 3000);
  };

  // ===== Callback accept =====
  const handleAccept = async (requestAddress: string) => {
    try {
      const currentAddress = address || (await connectWallet());
      const contract = await getContract();
      const tx = await contract.acceptFriendRequest(requestAddress);
      await tx.wait();

      const provider = new ethers.BrowserProvider(window.ethereum);
      const signer = await provider.getSigner();
      const message = `Xác nhận kết bạn với ${requestAddress}`;
      const signature = await signer.signMessage(message);

      await axios.post(`${import.meta.env.VITE_BACKEND_URL}/api/friends/accept`, {
        currentAddress,
        friendAddress: requestAddress,
        signature,
      });

      showNotification(`✅ Đã xác nhận lời mời từ ${requestAddress}`);
      setRequests(prev => prev.filter(r => r.user.address !== requestAddress));
    } catch (err) {
      console.error(err);
      showNotification("❌ Lỗi khi xác nhận lời mời");
    }
  };

  // ===== Callback reject =====
  const handleReject = async (requestAddress: string) => {
    try {
      const currentAddress = address || (await connectWallet());
      const contract = await getContract();
      const tx = await contract.rejectFriendRequest(requestAddress);
      await tx.wait();

      const provider = new ethers.BrowserProvider(window.ethereum);
      const signer = await provider.getSigner();
      const message = `Từ chối lời mời kết bạn từ ${requestAddress}`;
      const signature = await signer.signMessage(message);

      await axios.post(`${import.meta.env.VITE_BACKEND_URL}/api/friends/reject`, {
        currentAddress,
        friendAddress: requestAddress,
        signature,
      });

      showNotification(`🚫 Đã từ chối lời mời từ ${requestAddress}`);
      setRequests(prev => prev.filter(r => r.user.address !== requestAddress));
    } catch (err) {
      console.error(err);
      showNotification("❌ Lỗi khi từ chối lời mời");
    }
  };

  // ===== Fetch pending requests =====
  useEffect(() => {
    const fetchPending = async () => {
      if (!address) return;
      try {
        const res = await axios.get(
          `${import.meta.env.VITE_BACKEND_URL}/api/friends/${address}/pending`
        );
        if (res.data.success) {
          // map backend data => FriendRequest[]
          const mapped: FriendRequest[] = res.data.pending.map((r: any) => ({
            id: r.user2 || r.user1, // unique id
            user: {
              address: r.user1 === address ? r.user2 : r.user1,
              name: r.name || "Unknown",
              avatar: r.avatar || "",
              mutualFriends: r.mutualFriends || 0,
            },
          }));
          setRequests(mapped);
        } else {
          showNotification("Không thể tải lời mời kết bạn");
        }
      } catch (err) {
        console.error(err);
        showNotification("Lỗi khi gọi API lời mời");
      }
    };

    fetchPending();
  }, [address]);

  return (
    <div className="w-full p-6 bg-gray-50 min-h-screen">
      {notification && (
        <div className="fixed top-4 right-4 bg-green-600 text-white px-6 py-3 rounded-lg shadow-lg z-50">
          {notification}
        </div>
      )}

      <div className="max-w-7xl mx-auto">
        <div className="mb-6">
          <h1 className="text-gray-900 mb-2">Lời mời kết bạn</h1>
          <p className="text-gray-500">
            {requests.length > 0
              ? `${requests.length} lời mời kết bạn`
              : "Không có lời mời kết bạn mới"}
          </p>
        </div>

        <FriendRequestList
          requests={requests}
          onAccept={handleAccept}
          onReject={handleReject}
        />
      </div>
    </div>
  );
}
