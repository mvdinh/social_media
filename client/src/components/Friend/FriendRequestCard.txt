import { FriendRequest } from "../../types/friend";
import { AcceptButton } from "./AcceptButton";
import { RejectButton } from "./RejectButton";

interface FriendRequestCardProps {
  request: FriendRequest;
  onAccepted?: () => void;
  onRejected?: () => void;
}

export function FriendRequestCard({ request, onAccepted, onRejected }: FriendRequestCardProps) {
  return (
    <div className="bg-white border border-gray-200 rounded-lg overflow-hidden hover:shadow-lg transition-shadow">
      <img 
        src={request.user.avatar} 
        alt={request.user.name}
        className="w-full h-64 object-cover"
      />

      <div className="p-4">
        <h3 className="font-semibold text-gray-900 mb-1">{request.user.name}</h3>

        {request.user.mutualFriends && request.user.mutualFriends > 0 && (
          <div className="flex items-center gap-1 text-gray-500 text-sm mb-3">
            <div className="flex -space-x-2">
              <div className="w-4 h-4 rounded-full bg-gray-300 border border-white"></div>
              <div className="w-4 h-4 rounded-full bg-gray-400 border border-white"></div>
            </div>
            <span>{request.user.mutualFriends} bạn chung</span>
          </div>
        )}

        <div className="space-y-2">
          <AcceptButton requesterAddress={request.user.address} onAccepted={onAccepted} />
          <RejectButton requesterAddress={request.user.address} onRejected={onRejected} />
        </div>
      </div>
    </div>
  );
}
